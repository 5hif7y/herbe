cmake_minimum_required(VERSION 3.20)
project(herbe C)
set(CMAKE_C_STANDARD 11)
if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/config.h)
  configure_file(config.def.h ${CMAKE_CURRENT_SOURCE_DIR}/config.h COPYONLY)
endif()
add_executable(herbe herbe.c)
target_compile_definitions(herbe PRIVATE UNICODE _UNICODE)
target_compile_options(herbe PRIVATE $<$<C_COMPILER_ID:MSVC>:/W4 /utf-8>)
target_link_libraries(herbe PRIVATE user32 gdi32 shcore)
